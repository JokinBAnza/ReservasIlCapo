# ============================================================
# SOLO si el subdominio NO puede apuntar a la carpeta /public:
# subir este archivo a la RAIZ de la carpeta del subdominio
# con el nombre exacto:  .htaccess
# Redirige todas las visitas hacia /public, donde vive Laravel.
# ============================================================
<IfModule mod_rewrite.c>
    RewriteEngine On
    RewriteCond %{REQUEST_URI} !^/public/
    RewriteRule ^(.*)$ public/$1 [L]
</IfModule>
